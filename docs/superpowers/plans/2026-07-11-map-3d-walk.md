# 3D Farm Walk (Map Tab 2D ⇄ 3D) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a first-person 3D "walk" mode to the Map tab — walk forward with a joystick/WASD, drag to look around 360°, plants shown as health-colored markers at their real positions, tap to open the same plant sheet — toggleable back to the existing 2D Konva map.

**Architecture:** A new `components/farm/walk/` module renders a three.js scene via @react-three/fiber, client-only (next/dynamic, ssr:false), swapped with `FarmMapCanvas` inside the existing `FarmMap` wrapper by a `mode` state. Plant `position_x/position_y` (0–1 fractions of `farm.canvas_width/height`) map onto a ground plane; markers are THREE.Sprite billboards drawn with the same node style as the 2D `PlantMarker`. Camera is a walk rig: pointer-drag yaw/pitch + joystick/WASD movement clamped to the field bounds.

**Tech Stack:** Next.js 15, React 19, three + @react-three/fiber (v9, React-19 compatible), existing Konva 2D map untouched. No drei (sprites already billboard natively).

**Decisions locked with user (2026-07-11):** first-person walk (not photo spheres, not orbit view); lives as a 2D ⇄ 3D toggle on the Map tab; tapping a plant in 3D opens the same `PlantBottomSheet`; markers are billboard sprites in the 2D marker's visual language.

```ts
import type { Farm } from "@/types/database";

/**
 * World mapping — single source of truth for 2D↔3D coordinates.
 * The farm canvas (canvas_width × canvas_height px) becomes a ground plane
 * whose longest side is WORLD_LONG units; plant fractions (0–1) map onto it,
 * centered on the origin. +x = canvas right, +z = canvas down.
 */
const WORLD_LONG = 60;

export function worldSize(farm: Farm): { w: number; d: number } {
  const cw = farm.canvas_width || 1000;
  const ch = farm.canvas_height || 700;
  const s = WORLD_LONG / Math.max(cw, ch);
  return { w: cw * s, d: ch * s };
}

export function toWorld(
  farm: Farm,
  x01: number,
  y01: number
): [number, number, number] {
  const { w, d } = worldSize(farm);
  return [(x01 - 0.5) * w, 0, (y01 - 0.5) * d];
}

/** Eye height of the walking camera, in world units (~1.6 m). */
export const EYE_HEIGHT = 1.6;
```

- [ ] **Step 1: Create `greenmap/components/farm/walk/markerTexture.ts`**

Replicates the 2D node look (band-colored disc, white ring, bold % label) as a sprite texture. Colors are copied from `PlantMarker.tsx`'s `NODE_COLOR`.

```ts
import * as THREE from "three";
import type { HealthBand } from "@/lib/health/display";

const NODE_COLOR: Record<HealthBand, string> = {
  healthy: "#0DB074",
  attention: "#F59E0B",
  critical: "#EF4444",
  new: "#C7C7CC",
};

const cache = new Map<string, THREE.CanvasTexture>();

/** Draw the 2D marker style (colored disc + white ring + label) into a texture. */
export function markerTexture(band: HealthBand, label: string): THREE.CanvasTexture {
  const key = `${band}|${label}`;
  const hit = cache.get(key);
  if (hit) return hit;

  const S = 128;
  const canvas = document.createElement("canvas");
  canvas.width = S;
  canvas.height = S;
  const ctx = canvas.getContext("2d")!;
  const r = S / 2 - 8;

  ctx.beginPath();
  ctx.arc(S / 2, S / 2, r, 0, Math.PI * 2);
  ctx.fillStyle = NODE_COLOR[band];
  ctx.fill();
  ctx.lineWidth = 7;
  ctx.strokeStyle = "#ffffff";
  ctx.stroke();

  ctx.fillStyle = "#ffffff";
  ctx.font = `bold ${label.length > 3 ? 26 : 34}px -apple-system, Arial, sans-serif`;
  ctx.textAlign = "center";
  ctx.textBaseline = "middle";
  ctx.fillText(label, S / 2, S / 2 + 2);

  const tex = new THREE.CanvasTexture(canvas);
  tex.anisotropy = 2;
  cache.set(key, tex);
  return tex;
}
```

```tsx
"use client";

import { useMemo } from "react";
import type { ThreeEvent } from "@react-three/fiber";
import { healthDisplay } from "@/lib/health/display";
import type { Farm, Plant } from "@/types/database";
import { markerTexture } from "./markerTexture";
import { toWorld } from "./world";

const SPRITE_SIZE = 1.5; // world units — reads clearly from ~3–20 m away
const SPRITE_Y = 1.25; // floats at chest height like a field tag

/** One billboard sprite per active plant + a soft ground shadow disc. */
export function PlantSprites({
  farm,
  plants,
  onMarkerClick,
}: {
  farm: Farm;
  plants: Plant[];
  onMarkerClick: (plant: Plant) => void;
}) {
  const items = useMemo(
    () =>
      plants.map((p) => {
        const band = healthDisplay(p.health_score).band;
        const label = p.health_score === null ? "New" : `${p.health_score}%`;
        return {
          plant: p,
          pos: toWorld(farm, p.position_x, p.position_y),
          tex: markerTexture(band, label),
        };
      }),
    [farm, plants]
  );

  return (
    <group>
      {items.map(({ plant, pos, tex }) => (
        <group key={plant.id} position={pos}>
          <sprite
            position={[0, SPRITE_Y, 0]}
            scale={[SPRITE_SIZE, SPRITE_SIZE, 1]}
            onClick={(e: ThreeEvent<MouseEvent>) => {
              e.stopPropagation(); // don't also fire onPointerMissed
              onMarkerClick(plant);
            }}
          >
            <spriteMaterial map={tex} depthWrite={false} />
          </sprite>
          {/* soft shadow disc anchoring the sprite to the ground */}
          <mesh rotation={[-Math.PI / 2, 0, 0]} position={[0, 0.01, 0]}>
            <circleGeometry args={[0.45, 20]} />
            <meshBasicMaterial color="#1e2f1e" transparent opacity={0.25} />
          </mesh>
        </group>
      ))}
    </group>
  );
}
```

```tsx
"use client";

import { useEffect, useRef } from "react";
import { useFrame, useThree } from "@react-three/fiber";
import type { Farm } from "@/types/database";
import { worldSize, EYE_HEIGHT } from "./world";

const SPEED = 5; // world units / second (~brisk walk)
const LOOK_SPEED = 0.0045; // radians per pointer px
const PITCH_LIMIT = (75 * Math.PI) / 180;

/**
 * First-person rig: drag on the canvas to look (yaw/pitch), move via the
 * shared moveRef vector (keyboard here; joystick writes the same ref).
 * Position is clamped to the field with a 1-unit margin. No physics — YAGNI.
 */
export function WalkRig({
  farm,
  moveRef,
}: {
  farm: Farm;
  moveRef: React.RefObject<{ x: number; y: number }>;
}) {
  const { camera, gl } = useThree();
  const yaw = useRef(0); // yaw 0 faces -z — into the field from the spawn edge
  const pitch = useRef(0);
  const keys = useRef({ f: 0, b: 0, l: 0, r: 0 });

  // Pointer-drag look — attached to the WebGL canvas only, so the DOM
  // joystick (a sibling overlay) never fights it.
  useEffect(() => {
    const el = gl.domElement;
    let dragging = false;
    let lastX = 0;
    let lastY = 0;
    const down = (e: PointerEvent) => {
      dragging = true;
      lastX = e.clientX;
      lastY = e.clientY;
    };
    const move = (e: PointerEvent) => {
      if (!dragging) return;
      yaw.current -= (e.clientX - lastX) * LOOK_SPEED;
      pitch.current = Math.max(
        -PITCH_LIMIT,
        Math.min(PITCH_LIMIT, pitch.current - (e.clientY - lastY) * LOOK_SPEED)
      );
      lastX = e.clientX;
      lastY = e.clientY;
    };
    const up = () => {
      dragging = false;
    };
    el.addEventListener("pointerdown", down);
    window.addEventListener("pointermove", move);
    window.addEventListener("pointerup", up);
    return () => {
      el.removeEventListener("pointerdown", down);
      window.removeEventListener("pointermove", move);
      window.removeEventListener("pointerup", up);
    };
  }, [gl]);

  // WASD / arrow keys (desktop)
  useEffect(() => {
    const set = (e: KeyboardEvent, v: number) => {
      const k = e.key.toLowerCase();
      if (k === "w" || k === "arrowup") keys.current.f = v;
      else if (k === "s" || k === "arrowdown") keys.current.b = v;
      else if (k === "a" || k === "arrowleft") keys.current.l = v;
      else if (k === "d" || k === "arrowright") keys.current.r = v;
      else return;
      e.preventDefault();
    };
    const dn = (e: KeyboardEvent) => set(e, 1);
    const up = (e: KeyboardEvent) => set(e, 0);
    window.addEventListener("keydown", dn);
    window.addEventListener("keyup", up);
    return () => {
      window.removeEventListener("keydown", dn);
      window.removeEventListener("keyup", up);
    };
  }, []);

  useFrame((_, dt) => {
    const k = keys.current;
    const j = moveRef.current ?? { x: 0, y: 0 };
    const fwd = Math.max(-1, Math.min(1, k.f - k.b + j.y));
    const strafe = Math.max(-1, Math.min(1, k.r - k.l + j.x));
    const step = SPEED * Math.min(dt, 0.05); // clamp dt (tab-switch spikes)

    camera.position.x +=
      (Math.sin(yaw.current) * -fwd + Math.cos(yaw.current) * strafe) * step;
    camera.position.z +=
      (Math.cos(yaw.current) * -fwd - Math.sin(yaw.current) * strafe) * step;

    const { w, d } = worldSize(farm);
    camera.position.x = Math.max(-w / 2 + 1, Math.min(w / 2 - 1, camera.position.x));
    camera.position.z = Math.max(-d / 2 + 1, Math.min(d / 2 - 1, camera.position.z));
    camera.position.y = EYE_HEIGHT;
    camera.rotation.set(pitch.current, yaw.current, 0, "YXZ");
  });

  return null;
}
```

```tsx
"use client";

import { useEffect, useState } from "react";
import * as THREE from "three";
import type { Farm } from "@/types/database";
import { worldSize } from "./world";

/**
 * Field plane textured with the farm's uploaded/satellite photo when one
 * exists — the same image the 2D map shows, so 2D→3D feels like the same
 * place. Falls back to flat green (no texture) while loading or when absent.
 */
export function GroundTexture({ farm }: { farm: Farm }) {
  const [tex, setTex] = useState<THREE.Texture | null>(null);
  const { w, d } = worldSize(farm);

  useEffect(() => {
    const usable =
      farm.background_url &&
      (farm.background_type === "upload" || farm.background_type === "satellite");
    if (!usable) {
      setTex(null);
      return;
    }
    let disposed = false;
    const loader = new THREE.TextureLoader();
    loader.setCrossOrigin("anonymous");
    loader.load(farm.background_url!, (t) => {
      if (disposed) {
        t.dispose();
        return;
      }
      t.colorSpace = THREE.SRGBColorSpace;
      t.anisotropy = 4;
      setTex(t);
    });
    return () => {
      disposed = true;
      setTex((prev) => {
        prev?.dispose();
        return null;
      });
    };
  }, [farm.background_url, farm.background_type]);

  return (
    <mesh rotation={[-Math.PI / 2, 0, 0]} position={[0, 0, 0]}>
      <planeGeometry args={[w, d]} />
      {tex ? (
        <meshLambertMaterial map={tex} />
      ) : (
        <meshLambertMaterial color="#79a35c" />
      )}
    </mesh>
  );
}
```

- [ ] **Step 1: Create `greenmap/components/farm/walk/Joystick.tsx`**

```tsx
"use client";

import { useRef } from "react";

const SIZE = 104; // outer diameter px
const KNOB = 46;

/**
 * DOM joystick (bottom-left) writing a normalized {x,y} vector into moveRef
 * — read per-frame by WalkRig, so dragging never re-renders React. y is
 * +forward (up on the stick), x is +right.
 */
export function Joystick({
  moveRef,
}: {
  moveRef: React.RefObject<{ x: number; y: number }>;
}) {
  const baseRef = useRef<HTMLDivElement>(null);
  const knobRef = useRef<HTMLDivElement>(null);
  const active = useRef(false);

  function apply(clientX: number, clientY: number) {
    const base = baseRef.current;
    const knob = knobRef.current;
    if (!base || !knob) return;
    const rect = base.getBoundingClientRect();
    const max = (SIZE - KNOB) / 2;
    let dx = clientX - (rect.left + rect.width / 2);
    let dy = clientY - (rect.top + rect.height / 2);
    const len = Math.hypot(dx, dy);
    if (len > max) {
      dx = (dx / len) * max;
      dy = (dy / len) * max;
    }
    knob.style.transform = `translate(${dx}px, ${dy}px)`;
    if (moveRef.current) {
      moveRef.current.x = dx / max;
      moveRef.current.y = -dy / max; // screen-up = forward
    }
  }

  function reset() {
    active.current = false;
    if (knobRef.current) knobRef.current.style.transform = "translate(0px, 0px)";
    if (moveRef.current) {
      moveRef.current.x = 0;
      moveRef.current.y = 0;
    }
  }

  return (
    <div
      ref={baseRef}
      style={{ width: SIZE, height: SIZE, touchAction: "none" }}
      className="absolute bottom-4 left-4 z-10 rounded-full border-2 border-white/70 bg-black/15 backdrop-blur-sm"
      onPointerDown={(e) => {
        active.current = true;
        e.currentTarget.setPointerCapture(e.pointerId);
        apply(e.clientX, e.clientY);
      }}

      }}

      onPointerCancel={reset}
      aria-hidden
    >
      <div
        ref={knobRef}
        style={{ width: KNOB, height: KNOB }}
        className="absolute left-1/2 top-1/2 -ml-[23px] -mt-[23px] rounded-full bg-white/90 shadow-card"
      />
    </div>
  );
}
```
