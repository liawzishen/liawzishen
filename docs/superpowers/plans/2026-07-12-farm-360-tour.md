# Farm 360° Tour Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the synthetic first-person 3D walk with a Matterport-style photo-sphere tour: farmers upload real 360° photos pinned to spots on the 2D map, walk between spots via floor circles, and tag plants inside the photos (tap → existing PlantBottomSheet).

**Architecture:** A `components/farm/tour/` module renders the current panorama as an inverted three.js sphere (drag-to-look), computes hop-circle directions from the panoramas' 2D-map positions, and stores plant tags as yaw/pitch on the sphere. Data lives in two new Supabase tables + one storage bucket; `usePanoramas` (client hook, owned by FarmMap) is the single data-access layer. The 2D konva map doubles as the floor-plan view via 360° badges.

**Tech Stack:** Next.js 15 App Router, React 19, three ^0.185 + @react-three/fiber ^9.6 (already installed, lazy chunk), react-konva (2D), Supabase (Postgres + Storage, RLS), Tailwind v4, EN/BM dictionaries.

**Spec:** `docs/superpowers/specs/2026-07-12-farm-360-tour-design.md` (approved 2026-07-12)

Create `greenmap/components/farm/tour/usePanoramas.ts`:

```ts
"use client";

import { useCallback, useEffect, useState } from "react";
import { createClient } from "@/lib/supabase/client";
import type { FarmPanorama, PanoramaPlantTag } from "@/types/database";

/**
 * Single data-access layer for the 360° tour. Owned by FarmMap so both the
 * 2D badges and the 3D viewer share one fetch. A missing table (migration
 * 0009 not run yet) is treated as "no panoramas" + loadError — never a crash.
 */
export function usePanoramas(farmId: string) {
  const [panoramas, setPanoramas] = useState<FarmPanorama[]>([]);
  const [tags, setTags] = useState<PanoramaPlantTag[]>([]);
  const [loading, setLoading] = useState(true);
  const [loadError, setLoadError] = useState(false);

  useEffect(() => {
    let cancelled = false;
    async function load() {
      const supabase = createClient();
      const { data: panos, error: pErr } = await supabase
        .from("farm_panoramas")
        .select("*")
        .eq("farm_id", farmId)
        .order("created_at", { ascending: true });
      if (cancelled) return;
      if (pErr) {
        console.warn("[tour] panoramas fetch failed:", pErr.message);
        setLoadError(true);
        setLoading(false);
        return;
      }
      setPanoramas(panos ?? []);
      if ((panos ?? []).length > 0) {
        const { data: tagRows, error: tErr } = await supabase
          .from("panorama_plant_tags")
          .select("*")
          .in(
            "panorama_id",
            (panos ?? []).map((p) => p.id)
          );
        if (cancelled) return;
        if (tErr) console.warn("[tour] tags fetch failed:", tErr.message);
        setTags(tagRows ?? []);
      }
      setLoading(false);
    }
    load();
    return () => {
      cancelled = true;
    };
  }, [farmId]);

  const addPanorama = useCallback(
    async (input: { file: File; position_x: number; position_y: number }) => {
      const supabase = createClient();
      const ext = input.file.name.split(".").pop()?.toLowerCase() || "jpg";
      const path = `${farmId}/${crypto.randomUUID()}.${ext}`;
      const { error: upErr } = await supabase.storage
        .from("farm-panoramas")
        .upload(path, input.file, { contentType: input.file.type });
      if (upErr) {
        console.warn("[tour] upload failed:", upErr.message);
        return null;
      }
      const { data: pub } = supabase.storage
        .from("farm-panoramas")
        .getPublicUrl(path);
      const { data, error } = await supabase
        .from("farm_panoramas")
        .insert({
          farm_id: farmId,
          image_url: pub.publicUrl,
          position_x: input.position_x,
          position_y: input.position_y,
        })
        .select()
        .single();
      if (error || !data) {
        console.warn("[tour] insert failed:", error?.message);
        return null;
      }
      setPanoramas((prev) => [...prev, data as FarmPanorama]);
      return data as FarmPanorama;
    },
    [farmId]
  );

  const updateHeading = useCallback(async (id: string, heading_deg: number) => {
    setPanoramas((prev) =>
      prev.map((p) => (p.id === id ? { ...p, heading_deg } : p))
    );
    const supabase = createClient();
    const { error } = await supabase
      .from("farm_panoramas")
      .update({ heading_deg })
      .eq("id", id);
    if (error) console.warn("[tour] heading update failed:", error.message);
  }, []);

  const addTag = useCallback(
    async (input: {
      panorama_id: string;
      plant_id: string;
      yaw: number;
      pitch: number;
    }) => {
      const supabase = createClient();
      const { data, error } = await supabase
        .from("panorama_plant_tags")
        .upsert(input, { onConflict: "panorama_id,plant_id" })
        .select()
        .single();
      if (error || !data) {
        console.warn("[tour] tag insert failed:", error?.message);
        return null;
      }
      setTags((prev) => [
        ...prev.filter(
          (t) =>
            !(
              t.panorama_id === input.panorama_id &&
              t.plant_id === input.plant_id
            )
        ),
        data as PanoramaPlantTag,
      ]);
      return data as PanoramaPlantTag;
    },
    []
  );

  const removeTag = useCallback(async (id: string) => {
    setTags((prev) => prev.filter((t) => t.id !== id));
    const supabase = createClient();
    const { error } = await supabase
      .from("panorama_plant_tags")
      .delete()
      .eq("id", id);
    if (error) console.warn("[tour] tag delete failed:", error.message);
  }, []);

  return {
    panoramas,
    tags,
    loading,
    loadError,
    addPanorama,
    updateHeading,
    addTag,
    removeTag,
  };
}
```

Create `greenmap/components/farm/tour/geometry.ts`:

```ts
import type { Farm, FarmPanorama } from "@/types/database";

/** Radius of the photo sphere (camera sits at the origin inside it). */
export const SPHERE_R = 40;

/**
 * Yaw/pitch → position. Convention: yaw θ faces (-sin θ, 0, -cos θ), which is
 * exactly the direction of camera.rotation.set(pitch, yaw, 0, "YXZ").
 * pitch > 0 looks up. Positions sit just inside the sphere (r < SPHERE_R).
 */
export function toVec(
  yaw: number,
  pitch: number,
  r: number = SPHERE_R - 2
): [number, number, number] {
  return [
    -r * Math.cos(pitch) * Math.sin(yaw),
    r * Math.sin(pitch),
    -r * Math.cos(pitch) * Math.cos(yaw),
  ];
}

/** Inverse of toVec (for raycast hit points, in the SAME space as toVec). */
export function vecToSpherical(v: { x: number; y: number; z: number }): {
  yaw: number;
  pitch: number;
} {
  const r = Math.sqrt(v.x * v.x + v.y * v.y + v.z * v.z) || 1;
  return { yaw: Math.atan2(-v.x, -v.z), pitch: Math.asin(v.y / r) };
}

/**
 * Bearing from one panorama's standing spot to another on the 2D map,
 * in radians CLOCKWISE from map-north (map "up" = −z in tour space).
 * Fractions are scaled by the farm canvas so bearings respect its aspect.
 */
export function mapBearing(
  farm: Farm,
  from: FarmPanorama,
  to: FarmPanorama
): number {
  const dx = (to.position_x - from.position_x) * farm.canvas_width;
  const dy = (to.position_y - from.position_y) * farm.canvas_height;
  return Math.atan2(dx, -dy);
}

/** Map bearing → tour-space yaw (see toVec convention). */
export function bearingToYaw(bearing: number): number {
  return -bearing;
}
```

```tsx
"use client";

import { useEffect, useState } from "react";
import * as THREE from "three";
import type { FarmPanorama } from "@/types/database";
import { SPHERE_R } from "./geometry";

/**
 * Inverted sphere with the equirectangular photo on its inside. The mesh is
 * mirrored on X (equirect textures render mirrored from inside otherwise) and
 * rotated by heading_deg so the photo centre faces its real compass bearing.
 * Children (plant pins, Task 4) share the group so they stick to the photo.
 */
export function PanoramaSphere({
  pano,
  onLoaded,
  onLoadError,
  children,
}: {
  pano: FarmPanorama;
  onLoaded?: () => void;
  onLoadError?: () => void;
  children?: React.ReactNode;
}) {
  const [texture, setTexture] = useState<THREE.Texture | null>(null);

  useEffect(() => {
    let disposed = false;
    const loader = new THREE.TextureLoader();
    loader.load(
      pano.image_url,
      (tex) => {
        if (disposed) {
          tex.dispose();
          return;
        }
        tex.colorSpace = THREE.SRGBColorSpace;
        setTexture(tex);
        onLoaded?.();
      },
      undefined,
      () => {
        if (!disposed) onLoadError?.();
      }
    );
    return () => {
      disposed = true;
      setTexture((prev) => {
        prev?.dispose();
        return null;
      });
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [pano.image_url]);

  if (!texture) return null;

  return (
    <group rotation-y={THREE.MathUtils.degToRad(pano.heading_deg)}>
      <mesh scale={[-1, 1, 1]}>
        <sphereGeometry args={[SPHERE_R, 64, 48]} />
        <meshBasicMaterial map={texture} side={THREE.BackSide} />
      </mesh>
      {children}
    </group>
  );
}
```

```tsx
"use client";

import { useEffect, useRef } from "react";
import { useFrame, useThree } from "@react-three/fiber";

const LOOK_SPEED = 0.0045; // radians per pointer px
const PITCH_LIMIT = (75 * Math.PI) / 180;

/**
 * Drag to look around from the sphere centre. Tracks a single pointerId so
 * multi-touch can't cross streams; left button only; pitch clamped ±75°.
 */
export function LookControls() {
  const { camera, gl } = useThree();
  const yaw = useRef(0);
  const pitch = useRef(0);

  useEffect(() => {
    const el = gl.domElement;
    let activeId: number | null = null;
    let lastX = 0;
    let lastY = 0;
    const down = (e: PointerEvent) => {
      if (e.button !== 0 || activeId !== null) return;
      activeId = e.pointerId;
      lastX = e.clientX;
      lastY = e.clientY;
    };
    const move = (e: PointerEvent) => {
      if (e.pointerId !== activeId) return;
      yaw.current -= (e.clientX - lastX) * LOOK_SPEED;
      pitch.current = Math.max(
        -PITCH_LIMIT,
        Math.min(PITCH_LIMIT, pitch.current - (e.clientY - lastY) * LOOK_SPEED)
      );
      lastX = e.clientX;
      lastY = e.clientY;
    };
    const up = (e: PointerEvent) => {
      if (e.pointerId === activeId) activeId = null;
    };
    el.addEventListener("pointerdown", down);
    window.addEventListener("pointermove", move);
    window.addEventListener("pointerup", up);
    window.addEventListener("pointercancel", up);
    return () => {
      el.removeEventListener("pointerdown", down);
      window.removeEventListener("pointermove", move);
      window.removeEventListener("pointerup", up);
```

```tsx
"use client";

import { useMemo } from "react";
import type { ThreeEvent } from "@react-three/fiber";
import * as THREE from "three";
import type { Farm, FarmPanorama } from "@/types/database";
import { mapBearing, bearingToYaw, toVec } from "./geometry";

const CIRCLE_PITCH = -0.38; // radians below the horizon — reads as "on the ground"

/**
 * Matterport-style navigation: one semi-transparent white ring per other
 * panorama, placed at its real map bearing. Nearer spots render larger.
 * Rings live OUTSIDE the heading-rotated group — bearings are world-space.
 */
export function HopCircles({
  farm,
  current,
  panoramas,
  onHop,
}: {
  farm: Farm;
  current: FarmPanorama;
  panoramas: FarmPanorama[];
  onHop: (target: FarmPanorama) => void;
}) {
  const rings = useMemo(() => {
    const w = farm.canvas_width;
    const h = farm.canvas_height;
    return panoramas
      .filter((p) => p.id !== current.id)
      .map((p) => {
        const dx = (p.position_x - current.position_x) * w;
        const dy = (p.position_y - current.position_y) * h;
        const dist = Math.sqrt(dx * dx + dy * dy);
        const yaw = bearingToYaw(mapBearing(farm, current, p));
        // Nearer → bigger ring (clamped so far spots stay tappable).
        const scale = Math.max(1.4, Math.min(3.2, 900 / Math.max(dist, 1)));
        return { pano: p, pos: toVec(yaw, CIRCLE_PITCH), scale };
      });
  }, [farm, current, panoramas]);

  return (
    <group>
      {rings.map(({ pano, pos, scale }) => (
        <group key={pano.id} position={pos}>
          <mesh
            onClick={(e: ThreeEvent<MouseEvent>) => {
              e.stopPropagation();
              onHop(pano);
            }}
          >
            {/* invisible fat disc = ≥44px tap target */}
            <circleGeometry args={[scale * 0.9, 24]} />
            <meshBasicMaterial transparent opacity={0} depthWrite={false} />
          </mesh>
          <mesh>
            <ringGeometry args={[scale * 0.55, scale * 0.75, 32]} />
            <meshBasicMaterial
              color="#ffffff"
              transparent
              opacity={0.85}
              side={THREE.DoubleSide}
              depthWrite={false}
            />
          </mesh>
          <mesh>
            <circleGeometry args={[scale * 0.5, 24]} />
            <meshBasicMaterial
              color="#ffffff"
              transparent
              opacity={0.25}
              depthWrite={false}
            />
          </mesh>
        </group>
      ))}
    </group>
  );
}
```

```tsx
import { useMemo } from "react";
import type { ThreeEvent } from "@react-three/fiber";
import { healthDisplay } from "@/lib/health/display";
import type { PanoramaPlantTag, Plant } from "@/types/database";
import { markerTexture } from "./markerTexture";
import { toVec } from "./geometry";

const PIN_SIZE = 2.6;

/**
 * Matterport-style tags: one billboard sprite per tagged plant, positioned by
 * (yaw, pitch) in PHOTO space — render inside PanoramaSphere's group so pins
 * rotate with the photo when heading changes.
 */
export function PlantTagPins({
  tags,
  plants,
  editing,
  onPinClick,
}: {
  tags: PanoramaPlantTag[];
  plants: Plant[];
  editing: boolean;
  onPinClick: (tag: PanoramaPlantTag, plant: Plant) => void;
}) {
  const items = useMemo(
    () =>
      tags
        .map((tag) => {
          const plant = plants.find((p) => p.id === tag.plant_id);
          if (!plant) return null;

            tag,
            plant,
            pos: toVec(tag.yaw, tag.pitch),
            tex: markerTexture(band, label),
          };
```

```tsx
"use client";

import { useRef, useState } from "react";
import type { Farm } from "@/types/database";
import { useI18n } from "@/lib/i18n/LanguageProvider";

const MAX_BYTES = 12 * 1024 * 1024;

/**
 * Upload a 360° photo: pick file → aspect sanity check (warn, don't block)
 * → tap the mini-map where you stood → save. The mini-map is a plain div
 * (no konva) with the farm photo or the smart-background gradient.
 */
export function AddPanoramaSheet({
  farm,
  onSave,
  onClose,
}: {
  farm: Farm;
  onSave: (input: {
    file: File;
    position_x: number;
    position_y: number;
  }) => Promise<boolean>;
  onClose: () => void;
}) {
  const { t } = useI18n();
  const fileRef = useRef<HTMLInputElement>(null);
  const [file, setFile] = useState<File | null>(null);
  const [warn, setWarn] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [spot, setSpot] = useState<{ x: number; y: number } | null>(null);
  const [busy, setBusy] = useState(false);

  function pickFile(f: File | null) {
    setWarn(null);
    setError(null);
    if (!f) return;
    if (f.size > MAX_BYTES) {
      setError(t.tour.photoTooBig);
      return;
    }
    const url = URL.createObjectURL(f);
    const img = new Image();
    img.onload = () => {
      const ratio = img.naturalWidth / img.naturalHeight;
      if (Math.abs(ratio - 2) > 0.2) setWarn(t.tour.notPanorama);
      URL.revokeObjectURL(url);
    };
    img.src = url;
    setFile(f);
  }

  async function save() {
    if (!file || !spot || busy) return;
    setBusy(true);
    const ok = await onSave({ file, position_x: spot.x, position_y: spot.y });
    setBusy(false);
    if (!ok) setError(t.tour.loadFailed);
  }

  const hasBg =
    farm.background_url &&
    (farm.background_type === "upload" || farm.background_type === "satellite");

  return (
    <div className="fixed inset-0 z-50 flex items-end justify-center bg-black/45 lg:items-center">
      <div className="w-full max-w-[430px] rounded-t-card bg-card p-5 shadow-card lg:rounded-card">
        <p className="mb-3 text-[16px] font-semibold text-label">
          {t.tour.addPhoto}
        </p>

        <input
          ref={fileRef}
          type="file"
          accept="image/jpeg,image/png,image/webp"
          className="hidden"
          onChange={(e) => pickFile(e.target.files?.[0] ?? null)}
        />
        <button
          type="button"
          onClick={() => fileRef.current?.click()}
          className="pressable w-full rounded-card border border-dashed border-black/20 bg-fill px-4 py-3 text-[14px] font-medium text-secondary"
        >
          {file ? file.name : t.tour.addFirstHow}
        </button>
        <p className="mt-1.5 text-[11.5px] text-secondary">{t.tour.sizeHint}</p>

        {warn && <p className="mt-2 text-[12.5px] text-warning">{warn}</p>}
        {error && <p className="mt-2 text-[12.5px] text-danger">{error}</p>}

        {file && (
          <>
            <p className="mb-1.5 mt-4 text-[13.5px] font-medium text-secondary">
              {t.tour.whereStood}
            </p>
            <div
              className="relative w-full cursor-crosshair overflow-hidden rounded-card"
              style={{
                aspectRatio: `${farm.canvas_width} / ${farm.canvas_height}`,
                background: hasBg
                  ? undefined
                  : "linear-gradient(135deg, #DFF5E1, #C9EAD0)",
              }}
              onClick={(e) => {
                const r = e.currentTarget.getBoundingClientRect();
                setSpot({
                  x: Math.min(1, Math.max(0, (e.clientX - r.left) / r.width)),
                  y: Math.min(1, Math.max(0, (e.clientY - r.top) / r.height)),
                });
              }}
            >
              {hasBg && (
                // eslint-disable-next-line @next/next/no-img-element
                <img
                  src={farm.background_url!}
                  alt=""
                  className="absolute inset-0 h-full w-full object-cover"
                  draggable={false}
                />
              )}
              {spot && (
                <span
                  className="absolute h-4 w-4 -translate-x-1/2 -translate-y-1/2 rounded-full border-2 border-white bg-brand shadow-card"
                  style={{
                    left: `${spot.x * 100}%`,
                    top: `${spot.y * 100}%`,
                  }}
                />
              )}
            </div>
          </>
        )}

        <div className="mt-5 flex gap-3">
          <button
            type="button"
            onClick={onClose}
            className="pressable flex-1 rounded-card bg-fill px-4 py-2.5 text-[14px] font-semibold text-secondary"
          >
            {t.tour.cancel}
          </button>
          <button
            type="button"
            disabled={!file || !spot || busy}
            onClick={() => void save()}
            className="pressable flex-1 rounded-card bg-brand px-4 py-2.5 text-[14px] font-bold text-white disabled:opacity-50"
          >
            {busy ? t.tour.uploading : t.tour.save}
          </button>
        </div>
      </div>
    </div>
  );
}
```

```js
/**
 * Fetch 3 CC0 farmland panoramas (Poly Haven tonemapped JPGs, ~8K) and resize
 * to 2048×1024 q80 into public/demo/pano-{1,2,3}.jpg. If a download fails,
 * generate a stylized SVG panorama instead so the demo never breaks offline.
 * URL format verified 2026-07-12 (HTTP 200):
 *   https://dl.polyhaven.org/file/ph-assets/HDRIs/extra/Tonemapped%20JPG/<slug>.jpg
 * License: CC0 (https://polyhaven.com/license) — no attribution required.
 *
 * Run: node scripts/fetch-demo-panos.mjs
 */
import { writeFile, mkdir } from "node:fs/promises";
import { resolve } from "node:path";
import sharp from "sharp";

const SLUGS = ["flower_road", "meadow_2", "sunflowers_puresky", "hilly_terrain_01", "kloppenheim_06"];
const OUT = resolve(process.cwd(), "public/demo");
const W = 2048;
const H = 1024;

function fallbackSvg(i) {
  const skies = [["#BFE3FF", "#EAF6FF"], ["#FFD9A0", "#FFF3DC"], ["#C9E8FF", "#F2FAFF"]];
  const [top, bottom] = skies[i % skies.length];
  const rows = Array.from({ length: 7 }, (_, r) => {
    const y = H * 0.55 + r * (H * 0.06);
    return `<rect x="0" y="${y}" width="${W}" height="${H * 0.028}" fill="#5c7d4a" opacity="0.5" rx="8"/>`;
  }).join("");
  return `<svg xmlns="http://www.w3.org/2000/svg" width="${W}" height="${H}">
    <defs><linearGradient id="sky" x1="0" y1="0" x2="0" y2="1">
      <stop offset="0" stop-color="${top}"/><stop offset="1" stop-color="${bottom}"/>
    </linearGradient></defs>
    <rect width="${W}" height="${H * 0.55}" fill="url(#sky)"/>
    <rect y="${H * 0.55}" width="${W}" height="${H * 0.45}" fill="#79a35c"/>
    <circle cx="${W * 0.78}" cy="${H * 0.18}" r="46" fill="#F5D77A"/>
    ${rows}
  </svg>`;
}

async function fetchSlug(slug) {
  const url = `https://dl.polyhaven.org/file/ph-assets/HDRIs/extra/Tonemapped%20JPG/${slug}.jpg`;
  const res = await fetch(url);
  if (!res.ok) throw new Error(`${slug}: HTTP ${res.status}`);
  return Buffer.from(await res.arrayBuffer());
}

await mkdir(OUT, { recursive: true });
let written = 0;
for (const slug of SLUGS) {
  if (written >= 3) break;
  try {
    const raw = await fetchSlug(slug);
    const jpg = await sharp(raw).resize(W, H, { fit: "fill" }).jpeg({ quality: 80 }).toBuffer();
    await writeFile(resolve(OUT, `pano-${written + 1}.jpg`), jpg);
    console.log(`✓ pano-${written + 1}.jpg ← ${slug} (${(jpg.length / 1024) | 0} KB)`);
    written++;
  } catch (err) {
    console.warn(`⚠ ${slug} failed: ${err.message}`);
  }
}
while (written < 3) {
  const jpg = await sharp(Buffer.from(fallbackSvg(written))).jpeg({ quality: 80 }).toBuffer();
  await writeFile(resolve(OUT, `pano-${written + 1}.jpg`), jpg);
  console.log(`✓ pano-${written + 1}.jpg ← generated fallback`);
  written++;
}
console.log("Done — 3 demo panoramas in public/demo/.");
```

Create `greenmap/scripts/seed-panoramas.ts`:

```ts
/**
 * Seed 3 demo panoramas (committed under public/demo/) onto the demo farm
 * "Ladang Hijau" with a few plant tags each. Idempotent: wipes the demo
 * farm's panoramas first. Requires migration 0009 and the same env as
 * seed-demo.ts (NEXT_PUBLIC_SUPABASE_URL + SUPABASE_SERVICE_ROLE_KEY).
 *
 * Run: npx tsx scripts/seed-panoramas.ts   (npm run seed:panoramas)
 */
import { createClient } from "@supabase/supabase-js";
import { readFileSync } from "node:fs";
import { resolve } from "node:path";

function loadEnv() {
  try {
    const text = readFileSync(resolve(process.cwd(), ".env.local"), "utf8");
    for (const line of text.split("\n")) {
      const m = line.match(/^\s*([A-Z0-9_]+)\s*=\s*(.*)\s*$/);
      if (m && !process.env[m[1]]) {
        process.env[m[1]] = m[2].replace(/^["']|["']$/g, "");
      }
    }
  } catch {
    // no .env.local — rely on real environment variables
  }
}
loadEnv();

const SUPABASE_URL = process.env.NEXT_PUBLIC_SUPABASE_URL;
const SERVICE_KEY = process.env.SUPABASE_SERVICE_ROLE_KEY;
if (!SUPABASE_URL || !SERVICE_KEY) {
  console.error("✖ Missing NEXT_PUBLIC_SUPABASE_URL / SUPABASE_SERVICE_ROLE_KEY in .env.local");
  process.exit(1);
}
const admin = createClient(SUPABASE_URL, SERVICE_KEY, {
  auth: { autoRefreshToken: false, persistSession: false },
});

const DEMO_EMAIL = "ahmad@greenmap.demo";

// Standing spots + headings chosen to spread across the demo canvas.
const PANOS = [
  { file: "/demo/pano-1.jpg", x: 0.22, y: 0.3, heading: 0, label: "Near the gate" },
  { file: "/demo/pano-2.jpg", x: 0.55, y: 0.58, heading: 40, label: "Middle rows" },
  { file: "/demo/pano-3.jpg", x: 0.82, y: 0.32, heading: -60, label: "Far corner" },
];

async function main() {
  // 1. demo user → farm
  const { data: userList, error: uErr } = await admin.auth.admin.listUsers();
  if (uErr) throw uErr;
  const user = userList.users.find((u) => u.email === DEMO_EMAIL);
  if (!user) throw new Error(`Demo user ${DEMO_EMAIL} not found — run npm run db:seed first`);

  const { data: farm, error: fErr } = await admin
    .from("farms")
    .select("id")
    .eq("user_id", user.id)
    .limit(1)
    .single();
  if (fErr || !farm) throw new Error("Demo farm not found — run npm run db:seed first");

  // 2. wipe + insert panoramas (tags cascade)
  await admin.from("farm_panoramas").delete().eq("farm_id", farm.id);
  const { data: created, error: iErr } = await admin
    .from("farm_panoramas")
    .insert(
      PANOS.map((p) => ({
        farm_id: farm.id,
        image_url: p.file,
        position_x: p.x,
        position_y: p.y,
        heading_deg: p.heading,
        label: p.label,
      }))
    )
    .select();
  if (iErr || !created) throw iErr ?? new Error("insert failed");

  // 3. tag the 2 plants nearest each spot (spread around the horizon)
  const { data: plants, error: plErr } = await admin
    .from("plants")
    .select("id, position_x, position_y")
    .eq("farm_id", farm.id)
    .eq("lifecycle_status", "active");
  if (plErr || !plants) throw plErr ?? new Error("plants fetch failed");

  const tags: {
    panorama_id: string;
    plant_id: string;
    yaw: number;
    pitch: number;
  }[] = [];
  for (const pano of created) {
    const nearest = [...plants]
      .sort(
        (a, b) =>
          Math.hypot(a.position_x - pano.position_x, a.position_y - pano.position_y) -
          Math.hypot(b.position_x - pano.position_x, b.position_y - pano.position_y)
      )
      .slice(0, 2);
    nearest.forEach((plant, i) => {
      tags.push({
        panorama_id: pano.id,
        plant_id: plant.id,
        yaw: i === 0 ? 0.6 : -1.1, // spread left/right of centre
        pitch: -0.12, // slightly below the horizon, like plants on the ground
      });
    });
  }
  if (tags.length > 0) {
    const { error: tErr } = await admin.from("panorama_plant_tags").insert(tags);
    if (tErr) throw tErr;
  }

  console.log(`✓ Seeded ${created.length} panoramas + ${tags.length} plant tags on the demo farm.`);
}

main().catch((err) => {
  console.error("✖", err.message ?? err);
  process.exit(1);
});
```
